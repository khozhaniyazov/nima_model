1. **Create cache.py** with:

    ```python
    import hashlib
    import json
    from pathlib import Path
    from config import OUTPUTS

    class RenderCache:
        """Cache for rendered videos by code hash."""
        
        def __init__(self, cache_dir: Path = OUTPUTS / ".cache" / "renders"):
            self.cache_dir = cache_dir
            self.cache_dir.mkdir(parents=True, exist_ok=True)
        
        def get_code_hash(self, code: str) -> str:
            return hashlib.sha256(code.encode()).hexdigest()[:16]
        
        def check(self, code: str) -> Path | None:
            h = self.get_code_hash(code)
            cached = self.cache_dir / f"{h}.mp4"
            return cached if cached.exists() else None
        
        def store(self, code: str, video_path: Path) -> Path:
            h = self.get_code_hash(code)
            cached = self.cache_dir / f"{h}.mp4"
            import shutil
            shutil.copy2(video_path, cached)
            return cached
        
        def invalidate(self, code: str):
            h = self.get_code_hash(code)
            cached = self.cache_dir / f"{h}.mp4"
            if cached.exists():
                cached.unlink()

    class PromptCache:
        """Cache for AI generation responses by prompt hash."""
        
        def __init__(self, cache_dir: Path = OUTPUTS / ".cache" / "prompts"):
            self.cache_dir = cache_dir
            self.cache_dir.mkdir(parents=True, exist_ok=True)
        
        def get_prompt_hash(self, prompt: str, domain: str = "", voiceover: bool = False) -> str:
            data = json.dumps({"prompt": prompt, "domain": domain, "voiceover": voiceover}, sort_keys=True)
            return hashlib.sha256(data.encode()).hexdigest()[:24]
        
        def check(self, prompt: str, domain: str = "", voiceover: bool = False) -> dict | None:
            h = self.get_prompt_hash(prompt, domain, voiceover)
            meta_file = self.cache_dir / f"{h}.meta"
            if meta_file.exists():
                with open(meta_file) as f:
                    return json.load(f)
            return None
        
        def store(self, prompt: str, result: dict, domain: str = "", voiceover: bool = False):
            h = self.get_prompt_hash(prompt, domain, voiceover)
            meta_file = self.cache_dir / f"{h}.meta"
            with open(meta_file, "w") as f:
                json.dump(result, f)
        
        def invalidate_all(self):
            for f in self.cache_dir.glob("*.meta"):
                f.unlink()

    render_cache = RenderCache()
    prompt_cache = PromptCache()
    ```

    PROMPT_CACHE_ENABLED = True

    ```

3. **Initialize cache directories** in app.py startup.

    from concurrent.futures import ThreadPoolExecutor, as_completed

    def validate_while_generating(code: str, job_id: str) -> list:
        """Run validation checks in parallel."""
        checks = [
            ("syntax", lambda: validate_python_syntax(code)),
            ("structure", lambda: validate_names_and_imports(code)),
            ("security", lambda: validate_no_forbidden_calls(code)),
        ]
        
        results = []
        with ThreadPoolExecutor(max_workers=3) as executor:
            futures = {executor.submit(fn): name for name, fn in checks}
            for future in as_completed(futures):
                name = futures[future]
                try:
                    result = future.result()
                    if result:
                        results.extend(result)
                except Exception as e:
                    print(f"[VALIDATION] {name} error: {e}")
        
        return results
    ```
