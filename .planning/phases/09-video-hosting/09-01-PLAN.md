    ```python
    class VideoStorage:
        """Organized video storage with metadata tracking."""
        
        def __init__(self, outputs_dir: Path):
            self.outputs_dir = outputs_dir
        
        def _get_date_dir(self) -> str:
            """Return YYYY-MM-DD format for today's date."""
            return datetime.now().strftime("%Y-%m-%d")
        
        def _get_domain_dir(self, domain: str) -> str:
            """Return sanitized domain directory name."""
            return domain.lower().replace(" ", "_").replace("/", "_")
        
        def get_organized_path(self, job_id: str, domain: str = "general") -> Path:
            """
            Get organized storage path: outputs/{date}/{domain}/{job_id}.mp4
            
            Example: outputs/2026-04-05/math/sj8k2d.mp4
            """
            date_dir = self._get_date_dir()
            domain_dir = self._get_domain_dir(domain)
            return self.outputs_dir / date_dir / domain_dir / f"{job_id}.mp4"
        
        def organize_video(self, source_path: Path, job_id: str, domain: str = "general") -> Path:
            """
            Move/copy video to organized location.
            Creates directories as needed.
            """
            dest = self.get_organized_path(job_id, domain)
            dest.parent.mkdir(parents=True, exist_ok=True)
            
            if source_path.resolve() != dest.resolve():
                import shutil
                shutil.copy2(source_path, dest)
            
            return dest
        
        def get_video_info(self, job_id: str, domain: str = "general") -> dict:
            """Return video metadata dict for a given job_id."""
            organized_path = self.get_organized_path(job_id, domain)
            
            if not organized_path.exists():
                return None
            
            stat = organized_path.stat()
            return {
                "job_id": job_id,
                "organized_path": str(organized_path),
                "file_size_bytes": stat.st_size,
                "exists": True
            }
    
    # Initialize global VideoStorage instance
    video_storage = VideoStorage(OUTPUTS)
    ```

    ```python
    @app.route("/api/videos", methods=["GET"])
    def list_videos():
        """List videos with pagination and filtering."""
        if not db or not db.available:
            return jsonify({"error": "Database not available"}), 503
        
        page = request.args.get("page", 1, type=int)
        per_page = request.args.get("per_page", 20, type=int)
        domain = request.args.get("domain")
        sort_by = request.args.get("sort_by", "created_at")
        sort_order = request.args.get("sort_order", "desc")
        
        offset = (page - 1) * per_page
        order_col = "created_at" if sort_by == "created_at" else sort_by
        
        try:
            with db.conn.cursor(cursor_factory=RealDictCursor) as cur:
                # Build query
                base_query = """
                    SELECT v.*, r.prompt, r.domain, rj.status as render_status
                    FROM videos v
                    JOIN render_jobs rj ON v.render_job_id = rj.id
                    JOIN requests r ON v.request_id = r.id
                    WHERE 1=1
                """
                params = []
                
                if domain:
                    base_query += " AND r.domain = %s"
                    params.append(domain)
                
                # Get total count
                count_query = base_query.replace("SELECT v.*, r.prompt, r.domain, rj.status as render_status", "SELECT COUNT(*)")
                cur.execute(count_query, params)
                total = cur.fetchone()["count"]
                
                # Get page results
                base_query += f" ORDER BY v.{order_col} {'DESC' if sort_order == 'desc' else 'ASC'}"
                base_query += " LIMIT %s OFFSET %s"
                params.extend([per_page, offset])
                
                cur.execute(base_query, params)
                videos = cur.fetchall()
                
                return jsonify({
                    "videos": videos,
                    "pagination": {
                        "page": page,
                        "per_page": per_page,
                        "total": total,
                        "pages": (total + per_page - 1) // per_page
                    }
                })
        except Exception as e:
            return jsonify({"error": str(e)}), 500
    
    @app.route("/api/videos/search", methods=["GET"])
    def search_videos():
        """Search videos by prompt text."""
        if not db or not db.available:
            return jsonify({"error": "Database not available"}), 503
        
        query = request.args.get("q", "").strip()
        if not query:
            return jsonify({"videos": []})
        
        page = request.args.get("page", 1, type=int)
        per_page = request.args.get("per_page", 20, type=int)
        offset = (page - 1) * per_page
        
        try:
            with db.conn.cursor(cursor_factory=RealDictCursor) as cur:
                # Use full-text search on prompt
                cur.execute("""
                    SELECT v.*, r.prompt, r.domain, rj.status as render_status,
                           ts_rank(to_tsvector('english', r.prompt), plainto_tsquery('english', %s)) as rank
                    FROM videos v
                    JOIN render_jobs rj ON v.render_job_id = rj.id
                    JOIN requests r ON v.request_id = r.id
                    WHERE to_tsvector('english', r.prompt) @@ plainto_tsquery('english', %s)
                    ORDER BY rank DESC, v.created_at DESC
                    LIMIT %s OFFSET %s
                """, (query, query, per_page, offset))
                
                videos = cur.fetchall()
                
                return jsonify({
                    "videos": videos,
                    "query": query,
                    "count": len(videos)
                })
        except Exception as e:
            return jsonify({"error": str(e)}), 500
    
    @app.route("/api/videos/<video_id>", methods=["GET"])
    def get_video(video_id):
        """Get single video details by ID."""
        if not db or not db.available:
            return jsonify({"error": "Database not available"}), 503
        
        try:
            with db.conn.cursor(cursor_factory=RealDictCursor) as cur:
                cur.execute("""
                    SELECT v.*, r.prompt, r.domain, rj.status as render_status,
                           ae.overall_score, ae.visual_quality_score, ae.educational_value_score
                    FROM videos v
                    JOIN render_jobs rj ON v.render_job_id = rj.id
                    JOIN requests r ON v.request_id = r.id
                    LEFT JOIN ai_evaluations ae ON ae.render_job_id = rj.id
                    WHERE v.id = %s
                """, (video_id,))
                
                video = cur.fetchone()
                if not video:
                    return jsonify({"error": "Video not found"}), 404
                
                # Update last_accessed
                cur.execute("UPDATE videos SET last_accessed = CURRENT_TIMESTAMP WHERE id = %s", (video_id,))
                
                return jsonify(video)
        except Exception as e:
            return jsonify({"error": str(e)}), 500
    ```
