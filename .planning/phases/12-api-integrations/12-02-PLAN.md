    ```python
    # In-memory LTI session store (for OIDC flow)
    _lti_sessions: Dict[str, dict] = {}

    @app.route("/api/lti/login", methods=["GET", "POST"])
    def lti_login():
        """
        LTI 1.3 login initiation.
        Query params: iss (issuer), client_id, login_hint, target_link_uri
        """
        issuer = request.args.get("iss", "").strip()
        client_id = request.args.get("client_id", "").strip()
        login_hint = request.args.get("login_hint", "").strip()
        target_link_uri = request.args.get("target_link_uri", "").strip()
        
        if not all([issuer, client_id, login_hint, target_link_uri]):
            return jsonify({"error": "Missing required LTI parameters"}), 400
        
        # Look up platform
        if not db or not db.available:
            return jsonify({"error": "LTI not configured"}), 503
        
        platform = db.get_lti_platform_by_issuer(issuer)
        if not platform:
            return jsonify({"error": "Unknown LTI platform"}), 400
        
        # Create state and nonce for OIDC flow
        import secrets
        state = secrets.token_urlsafe(32)
        nonce = secrets.token_urlsafe(32)
        
        # Store session
        _lti_sessions[state] = {
            "issuer": issuer,
            "client_id": client_id,
            "platform": platform,
            "login_hint": login_hint,
            "target_link_uri": target_link_uri,
            "nonce": nonce
        }
        
        # Redirect to platform's auth endpoint
        auth_url = platform['auth_endpoint']
        params = {
            "response_type": "id_token",
            "response_mode": "form_post",
            "scope": "openid",
            "login_hint": login_hint,
            "nonce": nonce,
            "state": state,
            "redirect_uri": f"{request.url_root}api/lti/launch"
        }
        
        # Build auth URL
        from urllib.parse import urlencode
        full_url = f"{auth_url}?{urlencode(params)}"
        
        return jsonify({"auth_url": full_url, "state": state})

    @app.route("/api/lti/launch", methods=["POST"])
    def lti_launch():
        """
        LTI 1.3 launch handler.
        Receives id_token from platform, validates and extracts user context.
        """
        id_token = request.form.get("id_token", "").strip()
        state = request.form.get("state", "").strip()
        
        if not id_token or not state:
            return jsonify({"error": "Missing id_token or state"}), 400
        
        # Get session
        session = _lti_sessions.pop(state, None)
        if not session:
            return jsonify({"error": "Invalid state"}), 400
        
        # Decode and validate JWT (simplified - production should use proper JWT validation)
        try:
            import jwt
            # In production: verify with JWKS from platform
            claims = jwt.decode(id_token, options={"verify_signature": False})
            
            user_id = claims.get("sub", "")
            email = claims.get("email", "")
            name = claims.get("name", "")
            roles = claims.get("roles", [])
            
            # Check if instructor
            is_instructor = "Instructor" in roles or "ContentDeveloper" in roles
            
            # Create session for user
            lti_user_id = str(uuid.uuid4())
            
            return jsonify({
                "lti_user_id": lti_user_id,
                "user_id": user_id,
                "email": email,
                "name": name,
                "roles": roles,
                "is_instructor": is_instructor,
                "message": "LTI launch successful"
            })
            
        except Exception as e:
            return jsonify({"error": f"LTI launch failed: {str(e)}"}), 400
    ```

    ```python
    # Add to ManimDatabase class
    def save_batch(self, batch_id: str, total_jobs: int, webhook_url: str = None) -> str:
        """Save a new batch."""
        self._exec(
            """INSERT INTO batches (id, total_jobs, completed_jobs, failed_jobs, status, webhook_url)
               VALUES (%s, %s, %s, %s, %s, %s)
               ON CONFLICT (id) DO UPDATE SET
               total_jobs=EXCLUDED.total_jobs, webhook_url=EXCLUDED.webhook_url""",
            (batch_id, total_jobs, 0, 0, 'in_progress', webhook_url)
        )
        return batch_id

    def update_batch_progress(self, batch_id: str, completed: int = None,
                              failed: int = None, status: str = None):
        """Update batch progress counters."""
        if completed is not None:
            self._exec("UPDATE batches SET completed_jobs=%s WHERE id=%s",
                      (completed, batch_id))
        if failed is not None:
            self._exec("UPDATE batches SET failed_jobs=%s WHERE id=%s",
                      (failed, batch_id))
        if status:
            self._exec("UPDATE batches SET status=%s WHERE id=%s",
                      (status, batch_id))

    def get_batch_status(self, batch_id: str) -> dict | None:
        """Get batch status."""
        result = self._exec(
            """SELECT b.*, 
                      (SELECT COUNT(*) FROM render_jobs rj 
                       JOIN requests r ON rj.request_id = r.id
                       WHERE r.batch_id = %s AND rj.status = 'done') as completed,
                      (SELECT COUNT(*) FROM render_jobs rj 
                       JOIN requests r ON rj.request_id = r.id
                       WHERE r.batch_id = %s AND rj.status = 'error') as failed
               FROM batches b WHERE b.id = %s""",
            (batch_id, batch_id, batch_id), fetch='one'
        )
        return dict(result) if result else None
    ```

    2. **Add batch status endpoint:**

    ```python
    @app.route("/api/batch/<batch_id>", methods=["GET"])
    def api_batch_status(batch_id):
        """
        Get batch status and progress.
        Returns: {batch_id, total, completed, failed, in_progress, progress_percent, status}
        """
        # Count jobs in this batch from render_status
        batch_jobs = [
            (jid, status) for jid, status in render_status.items()
            if status.get("batch_id") == batch_id
        ]
        
        total = len(batch_jobs)
        completed = sum(1 for _, s in batch_jobs if s.get("status") == "done")
        failed = sum(1 for _, s in batch_jobs if s.get("status") == "error")
        in_progress = sum(1 for _, s in batch_jobs
                         if s.get("status") in ("generating", "rendering", "queued"))
        
        progress_percent = int((completed + failed) / total * 100) if total > 0 else 0
        
        status = "completed" if (completed + failed) >= total else "in_progress"
        if failed > 0 and (completed + failed) >= total:
            status = "completed_with_errors"
        
        return jsonify({
            "batch_id": batch_id,
            "total": total,
            "completed": completed,
            "failed": failed,
            "in_progress": in_progress,
            "progress_percent": progress_percent,
            "status": status
        })

    @app.route("/api/batch/<batch_id>/jobs", methods=["GET"])
    def api_batch_jobs(batch_id):
        """Get all jobs in a batch with their status."""
        batch_jobs = [
            {"job_id": jid, **status}
            for jid, status in render_status.items()
            if status.get("batch_id") == batch_id
        ]
        return jsonify({"batch_id": batch_id, "jobs": batch_jobs})
    ```
